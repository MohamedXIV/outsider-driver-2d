extends SceneTree
const OUT: String="res://work/clinic-main-frames"
var app: Node
var rows: Array=[]
func _initialize() -> void: call_deferred("_run")
func _pose(label: String) -> Dictionary:
	var w: SandboxWorld=app._world
	var row: Dictionary={"label":label,"minute":w.minute,"fraction":w.vehicle_fraction,"head":"e0ef24cfa0a8413cb76ef18060cc02a596c8f91c","world":w.snapshot(),"player_world":str(w.player_position(w.vehicle_fraction)),"player_render":str(SandboxDirectedLanes.taxi_screen(w,w.vehicle_fraction)),"bodies":app._vehicle_corridor.bodies.keys(),"parked":str(app._vehicle_corridor.parked.position) if app._vehicle_corridor.parked!=null else "","resident_count":w.residents.size()}
	if app._alight_view!=null:
		row["alight"]=app._alight_view.visual_snapshot()
	return row
func _shot(label: String) -> void:
	await RenderingServer.frame_post_draw
	var row: Dictionary=_pose(label)
	rows.append(row)
	if root.get_texture().get_image().save_png(OUT.path_join(label+".png"))!=OK: push_error("PNG capture failed");quit(1)
	FileAccess.open(OUT.path_join(label+".json"),FileAccess.WRITE).store_string(JSON.stringify(row,"  "))
	print("CLINIC_FRAME ",label," minute=",row["minute"])
func _run() -> void:
	DirAccess.make_dir_recursive_absolute(OUT)
	app=load("res://scenes/sandbox.tscn").instantiate()
	app._world=SandboxWorld.new(9)
	root.add_child(app)
	var offer: Dictionary={}
	for item in app._world.open_offers():
		if item["origin"]=="garage" and SandboxPassengerAlight.applicable(item): offer=item;break
	if offer.is_empty() or not app._world.accept(str(offer["id"])): push_error("No genuine medical offer");quit(1);return
	app._render()
	await _shot("01-departure")
	var approach: bool=false
	for frame in range(10800):
		await physics_frame
		if not approach and app._world.remaining_minutes<=2:
			approach=true;await _shot("02-approach")
		if not app._world.is_travelling(): break
	if app._world.is_travelling(): push_error("Real medical fare did not finish");quit(1);return
	await _shot("03-curb-and-rider")
	await create_timer(.3).timeout
	app._on_save()
	var saved: Dictionary=_pose("saved")
	app._on_load()
	var restored: Dictionary=_pose("restored")
	if saved.get("alight",{})!=restored.get("alight",{}) or saved["player_world"]!=restored["player_world"]:
		push_error("Save/Load actor pose changed");quit(1);return
	await _shot("04-restored")
	await create_timer(.5).timeout
	await _shot("05-walking")
	await create_timer(2.0).timeout
	await _shot("06-intake")
	var count: int=app._world.events.filter(func(e: Dictionary)->bool:return e["kind"]=="fare_settled" and e["subject"]==offer["id"]).size()
	if count!=1 or app._world.residents.size()!=9: push_error("Settlement/population changed");quit(1);return
	FileAccess.open(OUT.path_join("sequence.json"),FileAccess.WRITE).store_string(JSON.stringify(rows,"  "))
	print("CLINIC_MAIN_EVIDENCE_PASS genuine real-time fare, curb transfer, same saved poses, one settlement, nine residents")
	quit()
