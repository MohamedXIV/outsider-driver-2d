func _require(ok: bool,message: String) -> bool:
	if ok: return true
	push_error(message)
	JavaScriptBridge.eval("window.worldProofFailure="+JSON.stringify(message),true)
	return false
func _read(label: String) -> Dictionary:
	var w: SandboxWorld=app._world
	var bodies: Dictionary={}
	for id in app._vehicle_corridor.bodies:
		var body: SandboxWorldVehicleBody=app._vehicle_corridor.bodies[id]
		bodies[id]={"identity":body.identity,"x":body.position.x,"y":body.position.y,"heading":body.rotation}
	var point: Vector2=SandboxDirectedLanes.taxi_screen(w,w.vehicle_fraction)
	var people: Dictionary={}
	for npc in w.residents:
		var at: Vector2=SandboxVehicleDepartures.resident_screen(w,npc,w.vehicle_fraction)
		if at.is_finite(): people[npc["id"]]=[at.x,at.y]
	return {"head":HEAD,"mode":mode,"phase":label,"minute":w.minute,"fraction":w.vehicle_fraction,
		"taxi":[point.x,point.y],"bodies":bodies,"curb_residents":people,"population":w.residents.size(),"world":w.snapshot()}

					identity=app._vehicle_corridor.bodies[npc["id"]].identity;await _shot("approach");break
